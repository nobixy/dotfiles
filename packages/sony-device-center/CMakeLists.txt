cmake_minimum_required(VERSION 3.16)
project(SonyDeviceCenter VERSION 0.1.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SONY_REQUIRE_QT ON CACHE BOOL "" FORCE)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_subdirectory("${SONY_SOURCE}/libs/sony-transport" sony-transport)
add_subdirectory("${SONY_SOURCE}/libs/sony-protocol" sony-protocol)
add_subdirectory("${SONY_SOURCE}/libs/sony-core" sony-core)
add_subdirectory("${SONY_SOURCE}/apps/sonyd" sonyd)
add_subdirectory("${SONY_SOURCE}/apps/sonyctl" sonyctl)
add_subdirectory("${SONY_SOURCE}/apps/device-center" device-center)
